Why Eskiu

Drop a native module into your C++ stack.

Eskiu compiles to native code and speaks the C ABI directly. Take a memory-heavy C++ component, rewrite it as an explicit, auditable Eskiu module, and link it straight into the existing stack — you manage the memory, so the footprint is yours to control. The same language scales from a bare-metal kernel to an HTTP/2 service in the cloud.

Slots into your C/C++ stack

C ABI in and out, and it links as a plain native object. Call C libraries directly, hand Eskiu functions back as C callbacks, and drop a module into an existing codebase without rewriting the host.

Explicit memory, smaller footprint

You allocate and free in the source with alloc<T>(N) and free(ptr) — the memory is yours to account for, the lever teams pull when replacing C++. And 2.5× faster than reference C on our crypto benchmark.

Kernel to cloud, one language

Freestanding mode boots a bare-metal ARM64 kernel in QEMU. The same syntax and stdlib run an HTTP/2 service in the cloud. One language across the stack.

Examples

Looks like C.

Each file below is a complete program. Compile and run with eskiuc run file.esk.

async.esk 
extern int printf(string fmt, ...);
import <future>;
import <channel>;

// an async function compiles to a coroutine state machine.
async int sum_squares(Chan<int>* ch) {
    let a: int = await chan_recv(ch);
    let b: int = await chan_recv(ch);
    let c: int = await chan_recv(ch);
    return a + b + c;
}

int main() {
    Chan<int>* ch = chan_new<int>(8);
    for (i in 1..4) { chan_send(ch, i * i); }   // 1, 4, 9
    Future<int>* f = sum_squares(ch);
    future_poll<int>(f, void() {});             // drive it to completion
    printf("sum of squares = %d\n", f.value);   // 14
    free_future<int>(f);  chan_free<int>(ch);
    return 0;
}
// Output: sum of squares = 14
match.esk 
extern int printf(string fmt, ...);
extern int atoi(string s);

// Algebraic enum with payload variants.
enum ParseResult {
    Port(int),
    InvalidRange(int),
    NotANumber,
}

ParseResult parse_port(string s) {
    let n: int = atoi(s);
    if (n == 0) { return NotANumber; }
    if (n < 1 || n > 65535) { return InvalidRange(n); }
    return Port(n);
}

int main() {
    match parse_port("8080") {
        Port(p)         -> printf("Listening on :%d\n", p);
        InvalidRange(n) -> printf("%d out of range\n", n);
        NotANumber      -> printf("not a number\n");
    }
    return 0;
}
// Output: Listening on :8080
generics.esk 
extern int printf(string fmt, ...);

// A constraint: a type that can compare itself to another.
interface Ord { int cmp(Ord* o); }

// Coin satisfies Ord by providing cmp — structural, no "implements".
struct Coin {
    int value;
    int cmp(Coin* o) { if (self.value < o.value) { return -1; } return 1; }
}

// T is bounded by Ord; the compiler checks it at the call site.
T max<T: Ord>(T* a, T* b) {
    if (a.cmp(b) > 0) { return a[0]; }
    return b[0];
}

int main() {
    let x: Coin;  x.value = 7;
    let y: Coin;  y.value = 3;
    let hi: Coin = max<Coin>(&x, &y);
    printf("max = %d\n", hi.value);   // max = 7
    return 0;
}

import <list>;

int main() {
    let nums: List<int>;
    List_init(&nums, 4);
    for (i in 0..5) { List_push(&nums, i * i); }

    for (n in nums) { printf("%d\n", n); }   // 0 1 4 9 16
    return 0;

server.esk 
import <net>;
import <mem>;

int main() {
    int fd = net_tcp_listen(8080);
    while (1) {
        int c = net_accept(fd);
        *uint8 req = alloc<uint8>(4096);
        net_recv(c, (*void)req, 4096);
        net_send_str(c, "HTTP/1.1 200 OK\r\n\r\nHello from Eskiu.\n");
        free((*void)req);
        net_close(c);
    }
    return 0;
}
